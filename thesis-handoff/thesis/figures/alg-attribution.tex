\begin{algorithm}[t]
\caption{Scope attribution as a worklist fixpoint over $\mathcal{P}(S)$. The loops marked \emph{model only} belong to the full analysis and are not implemented in the
prototype evaluated in Chapter~\ref{ch:results}.}
\label{alg:attribution}
\begin{algorithmic}[1]
\Require routes $R$ with route tree, and the guards $\mathit{guards}(r)$ that route $r$ names
  in \texttt{canActivate} or \texttt{canMatch}; components $C$; nesting $N \subseteq C \times C$;
  HTTP interceptors $I$; call graph $\mathit{CG}$; dispatch relation $D$
\Ensure $\mathit{scope}(x) \in \mathcal{P}(S)$ for every component and callable $x$
\ForAll{$x$} $\mathit{scope}(x) \gets \varnothing$ \EndFor
\ForAll{$r \in R$ with $\mathit{component}(r) = c$}
  \State $\mathit{scope}(c) \gets \mathit{scope}(c) \cup \mathit{subtree}(r)$
  \Comment{own screen and outlet descendants}
\EndFor
\ForAll{$c$ on the nesting path from the root component to the holder of the primary outlet}
  \State $\mathit{scope}(c) \gets S$ \Comment{application shell, $\top$}
\EndFor
\ForAll{$r \in R$ and $g \in \mathit{guards}(r)$}
  \State $\mathit{scope}(g) \gets \mathit{scope}(g) \cup \mathit{subtree}(r)$
  \Comment{guard named on $r$}
\EndFor
\ForAll{HTTP interceptor $i \in I$} \Comment{\emph{model only}: registered for the application}
  \State $\mathit{scope}(i) \gets S$
\EndFor
\State $W \gets$ every $x$ with $\mathit{scope}(x) \neq \varnothing$
\While{$W \neq \emptyset$}
  \State take $x$ from $W$
  \ForAll{$y$ with $(x,y) \in N$} \Comment{$x$'s template renders $y$}
    \State \Call{Propagate}{$x, y$}
  \EndFor
  \ForAll{$y$ with $(x,y) \in \mathit{CG} \cup D$} \Comment{\emph{model only}: DI calls, dispatch}
    \State \Call{Propagate}{$x, y$}
  \EndFor
\EndWhile
\Procedure{Propagate}{$x, y$}
  \If{$\mathit{scope}(x) \not\subseteq \mathit{scope}(y)$}
    \State $\mathit{scope}(y) \gets \mathit{scope}(y) \cup \mathit{scope}(x)$; add $y$ to $W$
  \EndIf
\EndProcedure
\end{algorithmic}
\end{algorithm}
